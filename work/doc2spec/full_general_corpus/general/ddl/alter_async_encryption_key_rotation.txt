1.13.7.3 ALTER ASYNC ENCRYPTION KEY ROTATION

功能描述
           用于透明数据异步加密密钥轮转。
[[PDF_PAGE physical=1196 printed=1147]]
注意事项
       ●    仅在enable_tde与enable_tde_async_encryption参数开启时，该语句可被执行。
       ●    只有具备SYSADMIN权限的用户有权限执行此语句。
       ●    M-Compatibility数据库不支持该语句。
       ●    不支持记录统一审计日志。

语法格式
       ALTER ASYNC ENCRYPTION KEY ROTATION;




示例
       --确认透明加密异步加密开关打开。
       gaussdb=# show enable_tde_async_encryption;
        enable_tde_async_encryption
       -----------------------------
        on
       (1 row)

       --创建两张透明加密表t1、t2；执行\d元命令，可观察发现两张表使用同一个数据密钥（也即异步加密密钥）。
       gaussdb=# CREATE TABLE t1 (c1 INT, c2 TEXT) WITH (enable_tde = on);
       CREATE TABLE
       gaussdb=# CREATE TABLE t2 (c1 INT, c2 TEXT) WITH (enable_tde = on);
       CREATE TABLE
       gaussdb=# \d

       --轮转异步加密密钥。
       gaussdb=# ALTER ASYNC ENCRYPTION KEY ROTATION;
       ALTER TDE ROTATE ENCRYPTION KEY

       --创建新的透明加密表t3；执行\d元命令，可观察发现表t3数据密钥与t1、t2不同，也即异步加密密钥发生了实际
       轮转。
       gaussdb=# CREATE TABLE t3 (c1 INT, c2 TEXT) WITH (enable_tde = on);
       CREATE TABLE
       gaussdb=# \d

       --删除全部加密表。
       gaussdb=# DROP TABLE t1,t2,t3;
